cmake_minimum_required(VERSION 3.20)
project(CMotiveAndObjective VERSION 1.0.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(CMOTIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/toolchain/cmotive-0.3.0-selfhost" CACHE PATH "CMotive root containing lib/Sys/runtime.c")
find_program(CMOTIVE_COMPILER NAMES cmotive HINTS "${CMOTIVE_ROOT}/build/bin" REQUIRED)
if(NOT EXISTS "${CMOTIVE_ROOT}/lib/Sys/runtime.c")
 message(FATAL_ERROR "CMOTIVE_ROOT must contain lib/Sys/runtime.c")
endif()
file(GLOB_RECURSE CMO_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/CMotiveAndObjective/*.HMOT")
set(CMO_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/CMotiveAndObjective.c")
add_custom_command(OUTPUT "${CMO_GENERATED}"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/emit_core.py" --compiler "${CMOTIVE_COMPILER}" --compiler-root "${CMOTIVE_ROOT}" --output "${CMO_GENERATED}"
 DEPENDS VERSION ITERATION api.json src/CMotiveAndObjective.CMOT ${CMO_HEADERS} "${CMOTIVE_COMPILER}" tools/emit_core.py build-support/dependency-policy.json "${CMOTIVE_ROOT}/lib/Sys/runtime.c"
 WORKING_DIRECTORY "${CMOTIVE_ROOT}" VERBATIM)
add_library(CMotiveAndObjective SHARED "${CMO_GENERATED}")
target_compile_features(CMotiveAndObjective PRIVATE c_std_11)
target_include_directories(CMotiveAndObjective INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(CMotiveAndObjective PROPERTIES OUTPUT_NAME "CMotiveAndObjective-1.0.0")
if(WIN32)
 set_target_properties(CMotiveAndObjective PROPERTIES PREFIX "")
 target_sources(CMotiveAndObjective PRIVATE build-support/exports.def)
elseif(APPLE)
 target_link_options(CMotiveAndObjective PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/build-support/exports.macos")
 set_property(TARGET CMotiveAndObjective APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build-support/exports.macos")
else()
 target_link_options(CMotiveAndObjective PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/build-support/exports.map")
 set_property(TARGET CMotiveAndObjective APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build-support/exports.map")
endif()
if(UNIX)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_options(CMotiveAndObjective PRIVATE -fstack-protector-strong)
  target_link_options(CMotiveAndObjective PRIVATE "LINKER:--no-undefined,-z,relro,-z,now,-z,noexecstack")
 endif()
endif()
option(CMO_SANITIZE "Enable ASan and UBSan (GCC/Clang)" OFF)
if(CMO_SANITIZE AND NOT MSVC)
 target_compile_options(CMotiveAndObjective PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(CMotiveAndObjective PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
 if(UNIX)
  find_package(Threads REQUIRED)
 endif()
 add_executable(cmo_abi_test tests/abi_smoke.c)
 target_link_libraries(cmo_abi_test PRIVATE CMotiveAndObjective)
 if(CMO_SANITIZE AND NOT MSVC)
  target_compile_options(cmo_abi_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cmo_abi_test PRIVATE -fsanitize=address,undefined)
 endif()
 add_test(NAME abi COMMAND cmo_abi_test)
 add_executable(cmo_rb_invariants tests/rb_invariants.c)
 add_dependencies(cmo_rb_invariants CMotiveAndObjective)
 target_include_directories(cmo_rb_invariants PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 set(CMO_SMOKE_C "${CMAKE_CURRENT_BINARY_DIR}/cmotive-smoke.c")
 add_custom_command(OUTPUT "${CMO_SMOKE_C}"
  COMMAND "${CMOTIVE_COMPILER}" -I "${CMAKE_CURRENT_SOURCE_DIR}/include" --emit-c "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.CMOT" -o "${CMO_SMOKE_C}"
  DEPENDS tests/smoke.CMOT ${CMO_HEADERS} "${CMOTIVE_COMPILER}" VERBATIM)
 add_executable(cmo_source_test "${CMO_SMOKE_C}")
 target_include_directories(cmo_source_test PRIVATE "${CMOTIVE_ROOT}/lib/Sys")
 add_executable(cmo_unicode_exhaustive tests/unicode_exhaustive.c)
 add_dependencies(cmo_unicode_exhaustive CMotiveAndObjective)
 target_include_directories(cmo_unicode_exhaustive PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 set(CMO_NATIVE_FIXTURES cmo_rb_invariants cmo_source_test cmo_unicode_exhaustive)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(cmo_guard_pages tests/guard_pages.c)
  add_dependencies(cmo_guard_pages CMotiveAndObjective)
  target_include_directories(cmo_guard_pages PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  list(APPEND CMO_NATIVE_FIXTURES cmo_guard_pages)
  add_test(NAME guard_pages COMMAND cmo_guard_pages)
 endif()
 add_test(NAME unicode_exhaustive COMMAND cmo_unicode_exhaustive)
 foreach(fixture ${CMO_NATIVE_FIXTURES})
  target_compile_features(${fixture} PRIVATE c_std_11)
  if(fixture STREQUAL "cmo_source_test")
   if(WIN32)
    target_link_libraries(${fixture} PRIVATE ws2_32)
   else()
    target_link_libraries(${fixture} PRIVATE Threads::Threads m)
   endif()
  endif()
  if(CMO_SANITIZE AND NOT MSVC)
   target_compile_options(${fixture} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${fixture} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
 add_test(NAME rb_invariants COMMAND cmo_rb_invariants)
 add_test(NAME cmotive_source COMMAND cmo_source_test)
 if(NOT CMO_SANITIZE)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME core COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_core.py" $<TARGET_FILE:CMotiveAndObjective> "${CMAKE_CURRENT_BINARY_DIR}/test-result.json")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
   add_test(NAME dependencies COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_dependencies.py" $<TARGET_FILE:CMotiveAndObjective> --report "${CMAKE_CURRENT_BINARY_DIR}/dependency-report.json")
   add_test(NAME runtime_profile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runtime_profile.py" "${CMOTIVE_COMPILER}" "${CMOTIVE_ROOT}")
  endif()
  add_test(NAME release_reports COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_release_gates.py" $<TARGET_FILE:CMotiveAndObjective> "${CMAKE_CURRENT_BINARY_DIR}/test-result.json")
  set_tests_properties(release_reports PROPERTIES DEPENDS core)
  add_test(NAME metadata COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_metadata.py")
 endif()
endif()
install(TARGETS CMotiveAndObjective EXPORT CMotiveAndObjectiveTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
file(GLOB CMO_DOCS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/docs/*.md" "${CMAKE_CURRENT_SOURCE_DIR}/docs/*.json")
install(FILES LICENSE_1_0.txt toolchain/CMotive-LICENSE-MIT.txt ${CMO_DOCS} DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CMO_CONFIG_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/CMotiveAndObjective")
configure_package_config_file(cmake/CMotiveAndObjectiveConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/CMotiveAndObjectiveConfig.cmake" INSTALL_DESTINATION "${CMO_CONFIG_DIR}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/CMotiveAndObjectiveConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT CMotiveAndObjectiveTargets NAMESPACE CMotiveAndObjective:: DESTINATION "${CMO_CONFIG_DIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CMotiveAndObjectiveConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/CMotiveAndObjectiveConfigVersion.cmake" DESTINATION "${CMO_CONFIG_DIR}")
