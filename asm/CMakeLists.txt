cmake_minimum_required(VERSION 3.20)
project(CMotiveAndObjectiveRuntime VERSION 1.0.0 LANGUAGES NONE)
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(ZIG_EXECUTABLE NAMES zig python-zig REQUIRED)
get_filename_component(CMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CMO_ASM_OUT "${CMAKE_CURRENT_BINARY_DIR}/targets")
set(CMO_TARGETS x86-le x86_64-le arm32-le arm32-be arm64-le arm64-be)
file(GLOB_RECURSE CMO_ASM_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.s")
set(CMO_ASM_PRODUCTS)
foreach(cpu IN LISTS CMO_TARGETS)
 list(APPEND CMO_ASM_PRODUCTS "${CMO_ASM_OUT}/${cpu}/primitives.o" "${CMO_ASM_OUT}/${cpu}/primitives.elf" "${CMO_ASM_OUT}/${cpu}/libCMotiveAndObjectiveRuntime-1.0.0.a")
 install(FILES "${CMO_ASM_OUT}/${cpu}/libCMotiveAndObjectiveRuntime-1.0.0.a" DESTINATION "lib/CMotiveAndObjectiveRuntime/${cpu}")
endforeach()
add_custom_command(OUTPUT "${CMO_ASM_OUT}/build-report.json"
 BYPRODUCTS ${CMO_ASM_PRODUCTS}
 COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tools/build_asm.py" --zig "${ZIG_EXECUTABLE}" --output "${CMO_ASM_OUT}"
 DEPENDS ${CMO_ASM_SOURCES} contract.json "${CMO_ROOT}/tools/build_asm.py" VERBATIM)
add_custom_target(CMotiveAndObjectiveRuntime ALL DEPENDS "${CMO_ASM_OUT}/build-report.json")
if(BUILD_TESTING)
 add_test(NAME register_contract COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tests/test_asm.py" --build-dir "${CMO_ASM_OUT}")
 if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  add_test(NAME native_caller COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tests/test_asm_native.py" --build-dir "${CMO_ASM_OUT}")
 endif()
endif()
install(FILES README.md contract.json DESTINATION share/doc/CMotiveAndObjectiveRuntime)
