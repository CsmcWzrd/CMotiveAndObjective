cmake_minimum_required(VERSION 3.20)
project(CMotiveNativeQualification C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(CMO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(GEN "${CMO_ROOT}/prebuilt/generated")
add_library(CMotiveAndObjective SHARED "${GEN}/CMotiveAndObjective.c")
set_target_properties(CMotiveAndObjective PROPERTIES OUTPUT_NAME "CMotiveAndObjective-1.0.0")
if(MSVC)
 target_sources(CMotiveAndObjective PRIVATE "${CMO_ROOT}/build-support/exports.def")
 target_compile_options(CMotiveAndObjective PRIVATE /GS /guard:cf)
 target_link_options(CMotiveAndObjective PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
elseif(APPLE)
 target_compile_options(CMotiveAndObjective PRIVATE -fvisibility=hidden)
 target_link_options(CMotiveAndObjective PRIVATE "-Wl,-install_name,@rpath/libCMotiveAndObjective-1.0.0.dylib")
else()
 target_compile_options(CMotiveAndObjective PRIVATE -fPIC -fstack-protector-strong)
 target_link_options(CMotiveAndObjective PRIVATE "-Wl,--version-script=${CMO_ROOT}/build-support/exports.map" -Wl,-z,relro,-z,now,-z,noexecstack)
endif()
add_executable(portable_shared "${CMO_ROOT}/tests/replay_host.c")
target_compile_definitions(portable_shared PRIVATE CMO_DYNAMIC_TEST CMO_CONTAINER_TEST)
target_include_directories(portable_shared PRIVATE "${GEN}" "${CMO_ROOT}/tests" "${CMO_ROOT}/include")
target_link_libraries(portable_shared PRIVATE CMotiveAndObjective)
foreach(name rb_invariants unicode_exhaustive guard_pages)
 add_executable(${name} "${CMO_ROOT}/tests/${name}.c")
 target_include_directories(${name} PRIVATE "${GEN}")
endforeach()
enable_testing()
add_test(NAME portable_shared COMMAND portable_shared)
add_test(NAME rb_invariants COMMAND rb_invariants)
add_test(NAME unicode_exhaustive COMMAND unicode_exhaustive)
add_test(NAME guard_pages COMMAND guard_pages)
add_test(NAME core COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tests/test_core.py" "$<TARGET_FILE:CMotiveAndObjective>" "${CMAKE_BINARY_DIR}/test-result.json")
add_test(NAME metadata COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tests/check_metadata.py")
add_test(NAME reports COMMAND "${Python3_EXECUTABLE}" "${CMO_ROOT}/tests/test_release_gates.py" "$<TARGET_FILE:CMotiveAndObjective>" "${CMAKE_BINARY_DIR}/test-result.json")
set_tests_properties(reports PROPERTIES DEPENDS core)
